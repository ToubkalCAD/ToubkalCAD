  import type { ISketchSolver } from './ISketchSolver';
  import type { EntityGeom, DragPin, SolveResult } from './model';
  import type { SketchConstraint, SketchRef } from '../../store/cadStore';
  import { loadSolverModule, ExternalSystem, PointH, ParamH } from
  './externalSolver'; // your WASM glue

  export class ExternalSolverAdapter implements ISketchSolver {
    readonly id = 'planegcs';            // or 'solvespace'
    private mod!: Awaited<ReturnType<typeof loadSolverModule>>;

    async init() { this.mod = await loadSolverModule(); }

    solve(geoms: EntityGeom[], constraints: SketchConstraint[], dragPin?:
  DragPin): SolveResult {
      const sys = this.mod.createSystem();
      try {
        const reg = new HandleRegistry(sys);
        for (const g of geoms) reg.addEntity(g);            // (1) translate 
  geometry
        for (const c of constraints) addConstraint(sys, reg, c); // (2) register
  constraints
        if (dragPin) reg.applyDrag(dragPin);                // (3) drag 
  objective
        const status = sys.solve();                          // (3) solve
        return {
          geoms:      reg.readBack(geoms),                   // (4) coords back 
  into EntityGeom
          converged:  status.code === 'OK',
          residual:   status.residual ?? (status.code === 'OK' ? 0 : 1),
          iterations: status.iterations ?? 0,
        };
      } finally {
        sys.dispose();   // free WASM heap — same discipline as your WasmScope
      }
    }
  }

  (1)+(4) The translation layer — HandleRegistry

  This is where EntityGeom ↔ solver variables lives, and the only place that
  knows the external param model. It also resolves SketchRef → point handle,
  mirroring your pointIdx/pointCoords.

  type EntityH =
    | { kind: 'line';   a: PointH; b: PointH; ax: ParamH; ay: ParamH; bx:
  ParamH; by: ParamH }
    | { kind: 'circle'; c: PointH; cx: ParamH; cy: ParamH; r: ParamH }
    | { kind: 'arc';    c: PointH; cx: ParamH; cy: ParamH; r: ParamH; a1:
  number; a2: number };

  class HandleRegistry {
    private byId = new Map<string, EntityH>();
    constructor(private sys: ExternalSystem) {}

    addEntity(g: EntityGeom) {
      if (g.kind === 'line') {
        const ax = this.sys.addParam(g.a[0]), ay = this.sys.addParam(g.a[1]);
        const bx = this.sys.addParam(g.b[0]), by = this.sys.addParam(g.b[1]);
        this.byId.set(g.id, {
          kind: 'line', ax, ay, bx, by,
          a: this.sys.addPoint(ax, ay), b: this.sys.addPoint(bx, by),
        });
      } else {
        const cx = this.sys.addParam(g.c[0]), cy = this.sys.addParam(g.c[1]);
        const r  = this.sys.addParam(g.r);
        const c  = this.sys.addPoint(cx, cy);
        this.byId.set(g.id, g.kind === 'arc'
          ? { kind: 'arc', cx, cy, r, c, a1: g.a1, a2: g.a2 }
          : { kind: 'circle', cx, cy, r, c });
      }
    }

    entity(id: string) { return this.byId.get(id); }

    /** SketchRef → solver point handle. Mirrors pointCoords incl. arc 
  endpoints. */
    point(ref: SketchRef): PointH | null {
      const e = this.byId.get(ref.id);
      if (!e) return null;
      if (e.kind === 'line') return ref.pt === 'b' ? e.b : e.a;          // 'a' 
  default
      if (ref.pt === 'c' || !ref.pt) return e.c;                         // 
  centre
      // arc endpoint 'a'/'b': solver needs a real point — use the lib's 
  arc-endpoint
      // handle if it has one, else synthesise from centre+r+static angle.
      return null; // see §8 (arcs)
    }

    /** Pin params at current value (FIXED entities, datums). */
    fix(id: string) {
      const e = this.byId.get(id); if (!e) return;
      if (e.kind === 'line') [e.ax, e.ay, e.bx, e.by].forEach(p =>
  this.sys.setFixed(p, true));
      else { this.sys.setFixed(e.cx, true); this.sys.setFixed(e.cy, true);
  this.sys.setFixed(e.r, true); }
    }

    /** Drag: seed the dragged point at the cursor and mark it the solver's
     *  driven variable (SolveSpace: drag list; planegcs: weak point-on-point). 
  */
    applyDrag(pin: DragPin) {
      const p = this.point(pin.ref); if (!p) return;
      this.sys.setDragged(p, pin.target[0], pin.target[1]);
    }

    /** Read solved param values back into fresh EntityGeom (arc a1/a2 
  preserved). */
    readBack(geoms: EntityGeom[]): Record<string, EntityGeom> {
      const out: Record<string, EntityGeom> = {};
      for (const g of geoms) {
        const e = this.byId.get(g.id)!; const v = this.sys.getParam;
        out[g.id] = e.kind === 'line'
          ? { id: g.id, kind: 'line', a: [v(e.ax), v(e.ay)], b: [v(e.bx),
  v(e.by)] }
          : g.kind === 'arc'
            ? { id: g.id, kind: 'arc', c: [v(e.cx), v(e.cy)], r: Math.max(1e-4,
  v(e.r)), a1: g.a1, a2: g.a2 }
            : { id: g.id, kind: 'circle', c: [v(e.cx), v(e.cy)], r:
  Math.max(1e-4, v(e.r)) };
      }
      return out;
    }
  }

  readBack returns the exact SolveResult.geoms shape your bridge already 
  consumes — so geomKey diffing and rebuildSketchEntity work unchanged.

  (2) Constraint registration

  One pure function, SketchConstraint → native calls. Your refs-ordering and
  operand model carry straight over:

  function addConstraint(sys: ExternalSystem, reg: HandleRegistry, c:
  SketchConstraint) {
    const E = (i: number) => reg.entity(c.refs[i]?.id);
    const P = (i: number) => reg.point(c.refs[i]);

    switch (c.type) {
      case 'FIXED':       reg.fix(c.refs[0].id); break;
      case 'COINCIDENT':  { const a = P(0), b = P(1); if (a && b)
  sys.coincident(a, b); break; }
      case 'PARALLEL':    sys.parallel(E(0)!, E(1)!); break;
      case 'PERPENDICULAR': sys.perpendicular(E(0)!, E(1)!); break;
      case 'TANGENT':     sys.tangent(E(0)!, E(1)!); break;          // 
  line↔circle or circle↔circle
      case 'DISTANCE': {
        const a = P(0), b = P(1);
        if (a && b) sys.pointDistance(a, b, c.value!);              // 
  point–point
        else sys.pointLineDistance(P(0) ?? P(1)!, E(1) ?? E(0)!, c.value!); // 
  point–line
        break;
      }
      case 'HORIZONTAL':  sys.horizontal(E(0)!); break;
      case 'VERTICAL':    sys.vertical(E(0)!); break;
      case 'LENGTH':      sys.lineLength(E(0)!, c.value!); break;
      case 'RADIUS':      sys.radius(E(0)!, c.value!); break;
      case 'EQUAL':       sys.equal(E(0)!, E(1)!); break;
      case 'CONCENTRIC':  sys.coincident(reg.entity(c.refs[0].id)!.c,
  reg.entity(c.refs[1].id)!.c); break;
      case 'COLLINEAR':   sys.collinear(E(0)!, E(1)!); break;
      case 'ANGLE':       sys.angle(E(0)!, E(1)!, c.value! * Math.PI / 180);
  break; // your value is degrees
      case 'SYMMETRY':    sys.symmetric(P(0)!, P(1)!, E(2)!); break;
    }
  }

  Keep value units explicit at this boundary: your DISTANCE/LENGTH/RADIUS are
  mm, ANGLE is degrees (see cadStore.ts:118). Convert here, never inside the
  solver.

  6. Drag performance — the session fast-path

  Your liveSolve rebuilds the entire system every rAF frame
  (useCADConstraintPick.ts:273). The legacy solver tolerates it; a WASM solver
  will too for small sketches, but the right pattern is build-once-on-down:

  // in useCADConstraintPick onDown, after drag.ref is found:
  const solver = getSolver();
  drag.session = solver.beginSession?.(collectSolverGeoms(sid),
    [...persistedConstraints(sid), ...datumFixedConstraints()]) ?? null;

  // in solveStep:
  const res = drag.session
    ? drag.session.drag(drag.ref, [local.x, local.y])         // fast path
    : getSolver().solve([...real, ...datumGeoms()], cons, { ref: drag.ref,
  target: [local.x, local.y] });

  // in endDrag:
  drag.session?.dispose(); drag.session = null;

  beginSession is optional, so the legacy adapter can skip it and fall through
  to solve(). The external adapter implements it by keeping the ExternalSystem +
  HandleRegistry alive across the drag and only calling setDragged + solve +
  readBack per frame. This is where a WASM solver pays off most.

  ---
  8. Gotchas specific to your codebase

  - Datums stay external to the solver. Your
  datumGeoms()/datumFixedConstraints() inject FIXED axis lines at solve time.
  Keep doing exactly that — they arrive as ordinary EntityGeom + FIXED
  constraints and reg.fix() pins them. The solver never needs to know they're
  "special." isDatumId filtering in the bridge already skips them on read-back.
  - Arcs. Your model freezes a1/a2 and only varies [cx,cy,r]
  (SketchConstraintSolver.ts:32-39). Most external solvers model an arc as
  center + two endpoint params with their own DoF. Cheapest faithful mapping:
  register arcs as circles for center/radius constraints
  (Tangent/Concentric/Equal/Radius — exactly your current refOperand collapse to
  'circle'), and synthesize arc-endpoint refs in readBack from center + r + 
  static angle. If you later want arc endpoints to be real solver variables,
  that's a model upgrade, not a bridge change.
  - The anchor/rigidity tunables disappear. ANCHOR_W, RIGID_W, TAN_ARC_W, the
  multi-restart loop — those exist to coax your LM solver toward minimal-move,
  well-behaved solutions. SolveSpace and planegcs get minimal-move for free by
  seeding params at current values and regularizing the null-space internally.
  Don't port them; you'll fight the external solver's own dragging model. The
  arc-tangent-foot bias in particular is a workaround you can drop if the solver
  has a real arc-tangent constraint.
  - DoF / gating stays yours. computeDoF, constraintBlocked, canApply,
  CONSTRAINT_META are UI logic — leave them out of the solver entirely (move
  them to constraintModel.ts). Optionally cross-check computeDoF().dof against
  the solver's reported DoF as a sanity assertion in dev.
  - SolveResult.residual drives UI text (ConstraintPanel.tsx:163-165). External
  solvers report a status enum, not a Euclidean residual. Map OK→0,
  INCONSISTENT/REDUNDANT→1 (or the solver's reported error) so your "⚠ residual"
  message stays meaningful, or switch the message to the status code.

  Here's how JSketcher's solver actually works, which reshapes the translation
  layer:

  What JSketcher's API really looks like

  - System: new AlgNumSubSystem(calcVisualLimit, exprResolver, stage) →
  .addConstraint(c) → .solveFine() (or .solveRough()) → reads .solveStatus: { 
  success: boolean, error: number }.
  - A constraint is new AlgNumConstraint(schema, objects[], constants?) where
  schema is one of the named classes (PCoincident, Parallel, DistancePP,
  TangentLC, …) and objects are SolvableObjects — not raw params.
  - Geometry is objects that own Params. EndPoint has params: { x: Param, y: 
  Param }; a Param is new Param(value, sym), read with .get(), written with
  .set(). Segment owns two endpoints plus params: { ang, t } (angle + length);
  Circle/Arc own a center endpoint + radius param.

  The critical consequence: JSketcher constraints don't take coordinates — they 
  take its geometry objects, and lines are angle/length-parameterized.
  Parallel/Perpendicular/Horizontal/Vertical/Angle constrain segment.params.ang;
  SegmentLength constrains segment.params.t. Those derived params are kept
  consistent with the endpoints by the real Segment's own generative
  constraints. So the translation layer must instantiate JSketcher's shape 
  objects from your EntityGeom, not hand-build params.

  The good news: the ISketchSolver seam from before is unchanged. Only
  ExternalSolverAdapter → JSketcherSolverAdapter internals change.

  import type { EntityGeom, DragPin, SolveResult } from './model';
  import type { SketchConstraint, SketchRef } from '../../store/cadStore';

  // ── vendored JSketcher solver core (see "Packaging" below) ──
  import { AlgNumSubSystem } from '../../vendor/jsketcher/constr/AlgNumSystem';
  import { AlgNumConstraint } from
  '../../vendor/jsketcher/constr/ANConstraints';
  import {
    PCoincident, Parallel, Perpendicular, TangentLC,
    DistancePP, DistancePL, Horizontal, Vertical,
    SegmentLength, RadiusLength, EqualLength, EqualRadius,
    AngleBetween, Symmetry, LockPoint,
  } from '../../vendor/jsketcher/constr/ANConstraints';
  import { EndPoint } from '../../vendor/jsketcher/shapes/point';
  import { Segment } from '../../vendor/jsketcher/shapes/segment';
  import { Circle } from '../../vendor/jsketcher/shapes/circle';
  import { Arc } from '../../vendor/jsketcher/shapes/arc';

  export class JSketcherSolverAdapter implements ISketchSolver {
    readonly id = 'jsketcher';
    async init() {}                       // pure TS — nothing to load

    solve(geoms: EntityGeom[], constraints: SketchConstraint[], dragPin?:
  DragPin): SolveResult {
      const reg = new ShapeRegistry();
      const sys = new AlgNumSubSystem(() => 0, () => null, 0); // stage 0, no 
  expr resolver

      for (const g of geoms) reg.addEntity(g);
      for (const c of constraints) {
        for (const ac of toJSketcher(c, reg)) sys.addConstraint(ac);
      }
      let tempLock: AlgNumConstraint | null = null;
      if (dragPin) { tempLock = reg.lockAt(dragPin.ref, dragPin.target); if
  (tempLock) sys.addConstraint(tempLock); }

      sys.solveFine();
      const status = sys.solveStatus;

      return {
        geoms:      reg.readBack(geoms),
        converged:  status.success,
        residual:   status.error ?? (status.success ? 0 : 1),
        iterations: 0,                    // JSketcher doesn't surface an 
  iteration count
      };
    }

    // Drag fast-path: build the system once, swap only the temp lock per frame.
    beginSession(geoms: EntityGeom[], constraints: SketchConstraint[]):
  ISolveSession {
      const reg = new ShapeRegistry();

  class ShapeRegistry {
    private byId = new Map<string, Segment | Circle | Arc>();

    addEntity(g: EntityGeom) {
      if (g.kind === 'line') {
        this.byId.set(g.id, new Segment(new EndPoint(g.a[0], g.a[1]), new
  EndPoint(g.b[0], g.b[1])));
      } else if (g.kind === 'circle') {
        this.byId.set(g.id, new Circle(new EndPoint(g.c[0], g.c[1]), g.r));
      } else {
        // a1/a2 static in your model → build an Arc spanning them at the drawn 
  radius
        this.byId.set(g.id, new Arc(
          new EndPoint(g.c[0] + g.r * Math.cos(g.a1), g.c[1] + g.r *
  Math.sin(g.a1)),
          new EndPoint(g.c[0] + g.r * Math.cos(g.a2), g.c[1] + g.r *
  Math.sin(g.a2)),
          new EndPoint(g.c[0], g.c[1]),
        ));
      }
    }

    obj(id: string) { return this.byId.get(id)!; }

    /** SketchRef → JSketcher EndPoint (mirrors your pointCoords). */
    point(ref: SketchRef): EndPoint | null {
      const o = this.byId.get(ref.id); if (!o) return null;
      if (o instanceof Segment) return ref.pt === 'b' ? o.b : o.a;     // 'a' 
  default
      return (o as Circle | Arc).c;                                     // 
  centre
    }

    /** Temporary drag pin = LockPoint on the dragged point at the cursor. */
    lockAt(ref: SketchRef, t: [number, number]): AlgNumConstraint | null {
      const p = this.point(ref); if (!p) return null;
      p.x = t[0]; p.y = t[1];                       // seed so solver starts at 
  the cursor
      return new AlgNumConstraint(LockPoint, [p], [t[0], t[1]]);
    }

    /** Read solved endpoint params back into fresh EntityGeom — exactly the 
  shape
     *  your SketchSolveBridge.rebuildSketchEntity already consumes. */
    readBack(geoms: EntityGeom[]): Record<string, EntityGeom> {
      const out: Record<string, EntityGeom> = {};
      for (const g of geoms) {
        const o = this.byId.get(g.id)!;
        if (g.kind === 'line') {
          const s = o as Segment;
          out[g.id] = { id: g.id, kind: 'line', a: [s.a.x, s.a.y], b: [s.b.x,
  s.b.y] };
        } else {
          const c = o as Circle | Arc;
          out[g.id] = g.kind === 'arc'
            ? { id: g.id, kind: 'arc', c: [c.c.x, c.c.y], r: Math.max(1e-4,
  c.r.get()), a1: g.a1, a2: g.a2 }
            : { id: g.id, kind: 'circle', c: [c.c.x, c.c.y], r: Math.max(1e-4,
  c.r.get()) };
        }
      }
      return out;
    }
  }

  function toJSketcher(c: SketchConstraint, reg: ShapeRegistry):
  AlgNumConstraint[] {
    const O   = (i: number) => reg.obj(c.refs[i].id);
    const P   = (i: number) => reg.point(c.refs[i])!;
    const mk  = (schema: any, objs: any[], consts?: number[]) => [new
  AlgNumConstraint(schema, objs, consts)];
    const deg2rad = (d: number) => d * Math.PI / 180;

    switch (c.type) {
      case 'COINCIDENT':    return mk(PCoincident, [P(0), P(1)]);
      case 'CONCENTRIC':    return mk(PCoincident, [P(0), P(1)]);            // 
  centres coincide
      case 'PARALLEL':      return mk(Parallel,      [O(0), O(1)]);
      case 'PERPENDICULAR': return mk(Perpendicular, [O(0), O(1)]);
      case 'HORIZONTAL':    return mk(Horizontal,    [O(0)]);
      case 'VERTICAL':      return mk(Vertical,      [O(0)]);
      case 'LENGTH':        return mk(SegmentLength, [O(0)], [c.value!]);
      case 'RADIUS':        return mk(RadiusLength,  [O(0)], [c.value!]);
      case 'ANGLE':         return mk(AngleBetween,  [O(0), O(1)],
  [deg2rad(c.value!)]);
      case 'SYMMETRY':      return mk(Symmetry,      [P(0), P(1), O(2)]);
      case 'TANGENT': {
        const a = reg.obj(c.refs[0].id), b = reg.obj(c.refs[1].id);
        const seg = a instanceof Segment ? a : b, circ = a instanceof Segment ?
  b : a;
        if (seg instanceof Segment) return mk(TangentLC, [seg, circ]);       // 
  line ↔ circle/arc
        return ccTangent(a as Circle, b as Circle);                          // 
  circle ↔ circle (fallback, see notes)
      }
      case 'DISTANCE': {
        const p0 = reg.point(c.refs[0]), p1 = reg.point(c.refs[1]);
        if (p0 && p1) return mk(DistancePP, [p0, p1], [c.value!]);           // 
  point–point
        const pt = p0 ?? p1!, seg = (p0 ? reg.obj(c.refs[1].id) :
  reg.obj(c.refs[0].id)) as Segment;
        return mk(DistancePL, [pt, seg], [c.value!]);                        // 
  point–line
      }
      case 'EQUAL': {
        const a = reg.obj(c.refs[0].id), b = reg.obj(c.refs[1].id);
        return a instanceof Segment ? mk(EqualLength, [a, b]) : mk(EqualRadius,
  [a, b]);
      }
      case 'COLLINEAR':     return collinear(O(0) as Segment, O(1) as Segment);
  // see notes
      case 'FIXED':         return fixedEntity(c.refs[0].id, reg);
  // LockPoint per endpoint
      default:              return [];
    }
  }

  ---
  Why SolveSpace fits your stack far more cleanly than JSketcher

  The previous candidate failed because its solver was welded to its UI. libslvs
  is the opposite: it's the already-isolated solver core of SolveSpace, exposed
  through a stable C API (slvs.h) that has no notion of rendering. Three
  properties make it a near-native fit for ToubkalCAD:

  1. It works in workplane-local 2D. A SolveSpace POINT_IN_2D is two params (u, 
  v) inside a workplane — identical to your EntityGeom's local-2D model and your
  per-sketch Workplane. No impedance mismatch.
  2. Params are a flat array — literally the "raw parameter matrix" you want to
  feed in and read back.
  3. Units already match. Distances are unitless (use your mm directly), angles
  are in degrees (matches cadStore's value). Minimal conversion.

  And its group + dragged[] model maps directly onto two things you already
  have: fixed datums, and live-drag.

  Layer contract (the separation you asked for):
  - Zustand owns raw geometric parameters and constraints. Nothing else is
  authoritative.
  - The adapter is the only code that knows libslvs exists. It translates in,
  solves, translates out. Stateless between solves (or holds a session across a
  drag).
  - The solver (WASM) is pure math: params in, params out. No OCC, no Three, no
  store.
  - OCC + Three are downstream consumers of the solved Zustand state, reached
  through the existing bridge + CustomEvent bus. They never call the solver.

  The data crossing each boundary is plain JSON-able values (EntityGeom,
  SketchConstraint, number arrays) — no solver type ever leaks out of the
  adapter.

  ---
  Technical blueprint: param matrix → Solve() → store

  1. libslvs data model

  A Slvs_System is three flat arrays plus solve state:

  Slvs_Param      param[]       // { h, group, val }        ← the raw parameter 
  matrix
  Slvs_Entity     entity[]      // points, lines, circles, arcs, workplane 
  (reference param handles)
  Slvs_Constraint constraint[]  // { h, group, type, wrkpl, valA, ptA, ptB, 
  entityA..D }
  Slvs_hParam     dragged[4]    // params the solver keeps near current value 
  (drag)
  // outputs: int result (OKAY|INCONSISTENT|DIDNT_CONVERGE|TOO_MANY_UNKNOWNS), 
  int dof, hConstraint failed[]
  Slvs_Solve(sys, hGroup)       // solves params in hGroup; all other groups 
  held fixed

  Group strategy (this is the elegant part):
  - Group 1 = fixed reference: the sketch's workplane (origin point + normal),
  your datum origin/axes (SketchDatums), and any FIXED entities. Never solved.
  - Group 2 = active geometry: all free sketch entities. This is what
  Slvs_Solve(sys, 2) moves.

  Your existing datum concept is SolveSpace's fixed group — no soft-anchor hack
  needed.

  2. The adapter (SolveSpaceSolverAdapter)

  export class SolveSpaceSolverAdapter implements ISketchSolver {
    readonly id = 'solvespace';
    private mod!: SlvsModule;                 // the WASM module (Embind), 
  loaded once

    async init() { this.mod = await loadSlvs(); }   // like initOpenCascade → 
  window.oc

    solve(geoms, constraints, dragPin?): SolveResult {
      const sys = new this.mod.SketchSystem();      // C++ shim owns the 
  Slvs_System
      try {
        const reg = new HandleRegistry(sys);
        sys.addWorkplane(/* origin + normal from the sketch's Workplane, group 1
  */);
        for (const g of geoms)       reg.addEntity(g);             // (1) params
  + entities → matrix
        for (const c of constraints) addConstraint(sys, reg, c);   // (2) 
  constraints
        if (dragPin) reg.applyDrag(dragPin);                       // (3) 
  dragged[] + seed
        const r = sys.solve(/* group */ 2);                        // (3) 
  Slvs_Solve
        return {
          geoms:      reg.readBack(geoms),                         // (4) params
  → EntityGeom
          converged:  r.code === SLVS_RESULT_OKAY,
          residual:   r.code === SLVS_RESULT_OKAY ? 0 : 1,
          iterations: 0,
        };
      } finally { sys.delete(); }   // free WASM heap — same discipline as 
  WasmScope/registry
    }
  }

  HandleRegistry is the translation core — EntityGeom → param/entity handles and
  back, mirroring the role ShapeRegistry would have played:
  - line → 2 × POINT_IN_2D (4 params) + LINE_SEGMENT entity.
  - circle → POINT_IN_2D center (2 params) + DISTANCE entity for radius (1
  param) + CIRCLE.
  - arc → center point + start/end points + ARC_OF_CIRCLE (your static a1/a2 set
  the endpoint seeds; read back center+radius, preserve a1/a2).
  - point(ref) resolves a SketchRef → the entity's point handle (endpoint a/b,
  center c).
  - readBack pulls sys.getParam(h) for each handle into fresh EntityGeom — the 
  exact shape SketchSolveBridge.rebuildSketchEntity already consumes, so the
  bridge needs zero changes.

  4. Drag (replaces the legacy soft-anchor entirely)

  On pointer-down build the system once (the optional beginSession fast-path);
  each frame: set sys.dragged[] = the dragged point's two param handles, seed

  and the restart loop. The wiring in useCADConstraintPick
  (onDown/solveStep/endDrag) stays as-is; only what getSolver() returns changes.

  ---
  WASM packaging plan

  No maintained npm package exists — the official emscripten branch is stale and
  bundles the whole app. So build libslvs yourself, which is small and
  self-contained:

  1. Source: SolveSpace src/ solver files behind slvs.h (or the pre-isolated
  JacobStoren/SolveSpaceLib). No GUI, no OCC, no deps.
  2. Binding (recommended): a thin Embind C++ shim exposing a SketchSystem class
  (addWorkplane, addPoint2d, addLine, addCircle, addConstraint, setDragged,
  solve, getParam, delete). This keeps all Slvs_* struct/array marshalling in
  C++ — the JS adapter calls clean methods, never pokes HEAPF64. (The
  alternative, cwrap + manual heap marshalling of the three arrays, is workable
  but fiddly and error-prone.)
  3. Emscripten build: -O3, MODULARIZE=1, EXPORT_ES6=1, single-threaded (no
  pthreads → no SharedArrayBuffer needed). Output ~few-hundred-KB .wasm (vs
  OCC's 48 MB).
  4. Loading: loadSlvs() once at startup, parallel to initOpenCascade(). Stash
  on window.slvs (or keep it private to the adapter). Your existing COOP/COEP
  headers already cover it; SolveSpace doesn't require them since it's
  single-threaded, but they cause no conflict.
  5. Lifetime: the module is long-lived; each SketchSystem is created per solve
  (or per drag session) and .delete()d — same persistent-vs-temporary discipline
  as CADGeometryRegistry / WasmScope.

  Design (the important part)

  - Independent geometric checker, not self-report. maxViolation() re-derives
  each constraint's residual from solved coordinates (dot/cross/distance) and
  trusts neither solver. A solver "passes" only if its output genuinely
  satisfies the constraints — this is the adversarial-verify approach, and it's
  gauge-invariant, so under-constrained cases where the two solvers legitimately
  pick different solutions don't produce false failures.
  - Coord agreement only where it's meaningful. Cases pinned to the origin datum
  (compareCoords: true) have a unique solution, so the harness additionally
  asserts the two solvers' coordinates match (coordDiff).
  - Graceful degradation. Legacy runs always (pure TS). SolveSpace is loaded via
  a guarded dynamic import; until wasm/libslvs.mjs exists it reports SKIPPED 
  with the build command, never a false failure. The same call shape as the app
  (datums appended) is used for both.

  How it works, end to end

  1. Source of truth — Zustand. Each sketch entity is a sketch_wire node whose
  params.sketchGeom holds raw local-2D values (line {a,b}, circle {c,r}, arc 
  {c,r,a1,a2}). The constraint list lives on the sketch node's
  params.constraints.

  2. Trigger. Two entry points call the solver, both via getSolver().solve(...):
  - ConstraintPanel.solveAndApply — when you add/edit/delete a constraint or
  change a driving dimension.
  - useCADConstraintPick.liveSolve — every animation frame while you drag a
  point (passes a DragPin = {ref, target}).

  3. Translate. collectSolverGeoms(sketchId) reads the sketch's children into
  EntityGeom[], and the fixed datums (datumGeoms() + datumFixedConstraints() —
  the origin and U/V axes) are appended so geometry can be pinned to them.

  4. Solve (the LM math). solveConstraints builds a flat variable vector (line →
  4 vars a.u,a.v,b.u,b.v; circle/arc → 3 vars c.u,c.v,r), turns each constraint
  into residual equations, and minimizes ‖R‖² with a damped Gauss–Newton / 
  Levenberg–Marquardt iteration (finite-difference Jacobian). It adds:
  - a soft anchor so it picks the minimal-move solution,
  - rigidity anchors (hold a line's length / a circle's radius unless a
  dimension drives them),
  - multi-restart from perturbed seeds if a non-drag solve stalls.
  For a drag, the DragPin biases the dragged point toward the cursor while
  constraints still dominate. It returns { geoms, converged, residual }.

  5. Write back — Zustand → OCC → Three. For each entity whose geomKey changed,
  rebuildSketchEntity:
  - rebuilds the OpenCascade wire and re-registers it
  (CADGeometryRegistry.registerShape),
  - persists the new sketchGeom back to Zustand (setNodeParams),
  - fires cad-sketch-replace-visual, which Viewport3D/SketchDimensions listen
  for to redraw the Three.js polyline.
  Then propagateFromStore cascades the change to dependent features
  (extrude/revolve/loft built on the sketch).

  So: Zustand holds the params, the legacy LM solver moves them, the bridge 
  pushes results back into Zustand and repaints Three.js. SolveSpace will slot
  in at step 4 (same solve() signature) the moment you build the WASM and add
  the one-line installSolver(...) — no other code changes.

  To check at runtime: getSolver().id returns 'legacy-lm' today (it'll be
  'solvespace' once installed).

  How the activation works

  The wiring is conditional on the artifact being present, so it's a no-op today
  and switches on automatically once you build:

  1. loadSlvs.ts — imports the glue via import(/* webpackIgnore: true */ 
  './libslvs.mjs'). rspack resolves it at runtime only, so the app builds
  whether or not libslvs.mjs exists.
  2. build.sh — now SINGLE_FILE=1, so the (small) wasm is base64-embedded into
  one self-contained libslvs.mjs — no second .wasm fetch-path to get wrong.
  3. rspack.config.ts — a CopyRspackPlugin (noErrorOnMissing) drops libslvs.mjs
  into the dist root next to the bundle, exactly where the runtime import
  resolves. COOP/COEP and the OCC wasm rule are untouched.
  4. src/index.tsx — after kernel init, a guarded installSolver(new 
  SolveSpaceSolverAdapter()). If the glue loads, SolveSpace becomes active; if
  not, it logs and the legacy solver stays.

  The whole flow becomes: cd native/slvs && ./build.sh → rebuild/redeploy →
  SolveSpace is live. Console shows [solver] SolveSpace WASM active vs … legacy 
  solver active (or check getSolver().id). For npm run dev, restart the dev
  server after building so the copy is picked up.
